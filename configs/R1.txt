! R1 configuration
enable
configure terminal
hostname R1
no ip domain-lookup
enable secret Lab-Only-Pass1
service password-encryption
banner motd # Authorized access only #

line console 0
 password Lab-Only-Pass1
 login
 logging synchronous
exit

 description LAN1 to SW1
 ip address 192.168.5.190 255.255.255.192
 no shutdown
exit

 description LAN2 to SW2
 ip address 192.168.5.126 255.255.255.128
 no shutdown
exit

 description Link to R2

 no shutdown
exit
